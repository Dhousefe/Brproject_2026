🔐 Guia Rápido do Assinador HMAC-SHA256

Para consumir endpoints protegidos de qualquer linguagem externa, envie os headers calculados conforme a fórmula canônica:

METHOD | /raw/path | timestamp_millis | nonce_uuid | sha256_hex(raw_body_bytes) 
const crypto = require('crypto');

function signGameApi(secret, method, path, body = '') {
  const timestamp = Date.now().toString();
  const nonce = crypto.randomUUID();
  const bodyHash = crypto.createHash('sha256').update(body, 'utf8').digest('hex');
  const canonical = `${method.toUpperCase()}|${path}|${timestamp}|${nonce}|${bodyHash}`;
  const signature = crypto.createHmac('sha256', secret).update(canonical, 'utf8').digest('hex');

  return {
    'X-Site-Timestamp': timestamp,
    'X-Site-Nonce': nonce,
    'X-Site-Signature': signature,
    'Content-Type': 'application/json'
  };
}

import hmac, hashlib, uuid, time, requests

def sign_request(secret: bytes, method: str, path: str, body: str = ""):
    timestamp = str(int(time.time() * 1000))
    nonce = str(uuid.uuid4())
    body_hash = hashlib.sha256(body.encode('utf-8')).hexdigest()
    canonical = f"{method.upper()}|{path}|{timestamp}|{nonce}|{body_hash}"
    signature = hmac.new(secret, canonical.encode('utf-8'), hashlib.sha256).hexdigest()
    return {
        "X-Site-Timestamp": timestamp,
        "X-Site-Nonce": nonce,
        "X-Site-Signature": signature,
        "Content-Type": "application/json"
    }
